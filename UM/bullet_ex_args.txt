====================
BULLET EFFECTS
====================

EX_SPEEDUP: [UM = 0x1, UDoALG = 0x1, FW = 0 (broken)]
    No args. Broken in FW.

EX_ANIM: [UM = 0x2, UDoALG = 0x2, FW = 1]
    Runs interrupt 7 + A on the bullet ANM script

EX_ACCEL: [UM = 0x4, UDoALG = 0x4, FW = 2]
    duration = A
    acceleration = R

    S <= -999990.0                          : angle = current_angle
    S > -999990.0 && S < 999990.0           : angle = S
    S >= 999990.0 && S < 1999990.0          : angle = angle_to_player_from_current_position + M
    S >= 1999990.0 && S < 2999990.0         : angle = S
    S >= 2999990.0 && S < 3999990.0         : angle = angle_to_player_from_current_position + M * RAND_FLOAT_SIGNED
    S >= 3999990.0 && S < 4999990.0         : angle = current_angle + M * RAND_FLOAT_SIGNED
    S >= 4999990.0                          : angle = angle_from_boss0_to_current_position

EX_ANGLE_ACCEL: [UM = 0x8, UDoALG = 0x8, FW = 3]
    duration = A
    acceleration = R
    angle = S

EX_ANGLE: [UM = 0x10, UDoALG = 0x10, FW = 4]
    duration = a
    max_count = B
    type = C
    __int_44 = D

    S <= -999990.0  : effect_speed = current_speed
    S > -999990.0   : effect_speed = S
    C == 7          : effect_speed = current_speed + S * RAND_FLOAT_SIGNED

    R <= -999990.0                          : angle_arg = current_angle
    R > -999990.0 && R < 999990.0           : angle_arg = R
    R >= 999990.0 && R < 1999990.0          : angle_arg = angle_to_player_from_current_position + M
    R >= 1999990.0 && R < 2999990.0         : angle_arg = R
    R >= 2999990.0 && R < 3999990.0         : angle_arg = angle_to_player_from_current_position + M * RAND_FLOAT_SIGNED
    R >= 3999990.0 && R < 4999990.0         : angle_arg = current_angle + M * RAND_FLOAT_SIGNED
    R >= 4999990.0                          : angle_arg = angle_from_boss0_to_current_position

    C == 0 || C == 1 || C == 4              : effect_angle = angle_arg
    C == 2                                  : effect_angle = angle_to_player_from_saved_position + angle_arg
    C == 3                                  : effect_angle = saved_angle + angle_arg
    C == 5 || C == 6                        : effect_angle = R * RAND_FLOAT_SIGNED
    C == 7 && R <= -999990.0                : effect_angle = current_angle
    C == 7 && R > -999990.0 && R < 990.0    : effect_angle = R
    C == 7 && R >= 999.0                    : effect_angle = angle_to_player_from_current_position
    
When duration runs out:
    speed = effect_speed

    C == 0 || C == 5            : angle += effect_angle
    C == 1 || C == 6            : angle += effect_angle + angle_to_player_from_current_position
    C == 2 || C == 3 || C == 4  : angle = effect_angle

EX_BOUNCE: [UM = 0x40, UDoALG = 0x40, FW = 6]
    max_count = A
    type = B
    speed = R
    
    B & 0x10    : Don't actually bounce, only detect when the bullet leaves the range
    
    B & 0x20    : size_x = S, size_y = M
    !(B & 0x20) : size_x = 384.0, size_y = 448.0

EX_INVULN: [UM = 0x80, UDoALG = 0x80, FW = 7]
    duration = A

EX_OFFSCREEN: [UM = 0x100, UDoALG = 0x100, FW = 8]
    max_time = A
    
    B == 0      : Bullet cannot despawn offscreen
    B != 0      : Bullet will still despawn offscreen if it's not moving towards the screen

EX_SETSPRITE: [UM = 0x200, UDoALG = 0x200, FW = 9]
    sprite = A
    color = B & 0x7FFF

    B & 0x8000 : runs interrupt 2 on the bullet vm

EX_DELETE: [UM = 0x400, UDoALG = 0x400, FW = 10]
    A == 1 : Disable cancel ANM script

EX_PLAYSOUND: [UM = 0x800, UDoALG = 0x800, FW = 11]
    sound_id = A

EX_WRAP: [UM = 0x1000, UDoALG = 0x1000, FW = 12]
    max_count = A
    walls = B

EX_SHOOT: [UM = 0x2000, UDoALG = 0x2000, FW = 13]
    aim_mode = A
    effect_index = B
    count1 = C
    count2 = D
    sprite = A2
    color = B2
    delete_current_bullet = C2

    R <= -999990.0                          : angle = current_angle
    R > -999990.0 && R <= 999990.0          : angle = R
    R > 999990.0 && R < 1999990.0           : angle = angle_to_player_from_current_position
    R >= 1999990.0                          : angle = R

    angle2 = S

    M <= -999990.0  : speed1 = current_speed
    M > -999990.0   : speed1 = M
    
    speed2 = N

EX_SHOOT_DATA: [UM = 0x4000, UDoALG = 0x4000, FW = 14]
    Doesn't do anything on its own, but provides more args for EX_SHOOT

EX_REACT: [UM = 0x8000, UDoALG = 0x8000, FW = 15]
    __ex_react_val = A

EX_LOOP: [UM = 0x10000, UDoALG = 0x10000, FW = 16]
    effect_index = A
    
    B <= 0 : loop_count ignored, acts as EX_GOTO
    B > 0  : loop_count = B

EX_MOVE: [UM = 0x20000, UDoALG = 0x20000, FW = 17]
    duration = A
    interp_mode = B & 0xFF
    target_x = R
    target_y = S
    
    B & 0x100    : Target is absolute
    !(B & 0x100) : Target is relative to current position
    
EX_VEL: [UM = 0x40000, UDoALG = 0x40000, FW = 18]
    R >= 990.0                  : angle = angle_to_player_from_current_position + (R - 999.0)
    R >= -990.0 && R < 990.0    : angle = R
    R < -990.0                  : angle = current_angle
    
    S < -990.0  : speed = current_speed
    S >= -990.0 : speed = S

EX_POSADD: [UM = 0x80000, UDoALG = 0x80000, FW = 19]
    duration = A
    angle = R
    speed = S

EX_BLEND: [UM = 0x100000, UDoALG = 0x100000, FW = 20]
    A == 1              : blend_mode = BlendAdditive
    A == 2              : blend_mode = BlendSubtractive
    A != 1 && A != 2    : blend_mode = BlendNormal

EX_VELTIME: [UM = 0x200000, UDoALG = 0x200000, FW = 21]
    duration = A
    speed = (R - current_speed) / A

    S <= -999990.0                          : angle = current_angle
    S > -999990.0 && S < 999990.0           : angle = S
    S >= 999990.0 && S < 1999990.0          : angle = angle_to_player_from_current_position + M
    S >= 1999990.0 && S < 2999990.0         : angle = S
    S >= 2999990.0 && S < 3999990.0         : angle = angle_to_player_from_current_position + M * RAND_FLOAT_SIGNED
    S >= 3999990.0 && S < 4999990.0         : angle = current_angle + M * RAND_FLOAT_SIGNED
    S >= 4999990.0                          : angle = angle_from_boss0_to_current_position

EX_LASER: [UM = 0x8000000, UDoALG = 0x8000000, FW = 27]
    A == 0 : line laser

    sprite = B
    color = C
    delete_current_bullet = D
    shoot_sound = A2
    transform_sound = B2
    effect_index = C2

    R <= -999990.0                          : angle = current_angle
    R > -999990.0 && R <= 999990.0          : angle = R
    R > 999990.0 && R < 1999990.0           : angle = angle_to_player_from_current_position
    R >= 1999990.0                          : angle = R

    S <= -999990.0  : speed = current_speed
    S > -999990.0   : speed = S

    length = M
    max_length = N
    despawn_distance = R2
    width = S2
    spawn_distance = M2

    A == 1 : Infinite laser

    sprite = B
    color = C
    flags = (D & 0xFD) | 0x02
    effect_index = (D & 0xFF00) >> 8
    delete_current_bullet = (D & 0x10000) >> 16
    start_time = A2
    expand_time = B2
    duration = C2
    stop_time = D2
    shoot_sound = 18
    transform_sound = -1

    R <= -999990.0                          : angle = current_angle
    R > -999990.0 && R <= 999990.0          : angle = R
    R > 999990.0 && R < 1999990.0           : angle = angle_to_player_from_current_position
    R >= 1999990.0                          : angle = R

    S <= -999990.0  : speed = current_speed
    S > -999990.0   : speed = S

    length = M
    max_length = N
    width = R2
    distance = S2
